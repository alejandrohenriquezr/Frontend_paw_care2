        function filterByVet(vetId) {
            const grid = document.getElementById('calendar-grid');
            const header = document.getElementById('days-header');
            const vet = vetSchedules[vetId];
            
            // Update grid to single column
            grid.className = 'ml-16 grid grid-cols-1 transition-all duration-300 max-w-md mx-auto';
            header.className = 'grid grid-cols-1 border-b border-gray-200 max-w-md mx-auto';
            
            // Update header
            header.innerHTML = `
                <div class="p-4 text-center bg-pawcare-gray">
                    <p class="font-medium">${vet.name}</p>
                    <p class="text-sm text-gray-600">Lun - Dom</p>
                </div>
            `;
            
            // Generate time slots
            let slotsHTML = '';
            for (let hour = 8; hour <= 19; hour++) {
                const hasAppointment = vet.schedule.find(apt => 
                    apt.time === hour || (apt.duration && apt.time <= hour && apt.time + apt.duration > hour)
                );
                
                if (hasAppointment && hasAppointment.time === hour) {
                    const bgColor = hasAppointment.status === 'confirmed' ? 'bg-pawcare-green' : 'bg-pawcare-yellow';
                    const height = hasAppointment.duration ? `height: ${hasAppointment.duration * 64 + (hasAppointment.duration - 1) * 1}px;` : 'h-14';
                    
                    slotsHTML += `
                        <div class="h-16 border-b border-gray-200 relative">
                            <div class="absolute top-1 left-1 right-1 ${bgColor} rounded p-2 cursor-pointer" style="${hasAppointment.duration ? height : ''}" onclick="showAppointmentModal('${hasAppointment.appointment}')">
                                <p class="text-sm font-medium">${hasAppointment.appointment}</p>
                                <p class="text-xs text-gray-600">${vet.name}</p>
                            </div>
                        </div>
                    `;
                    
                    // Skip additional slots for multi-hour appointments
                    if (hasAppointment.duration) {
                        for (let i = 1; i < hasAppointment.duration; i++) {
                            slotsHTML += `<div class="h-16 border-b border-gray-200 relative"></div>`;
                            hour++;
                        }
                    }
                } else if (hasAppointment && hasAppointment.time < hour) {
                    // Part of multi-hour appointment
                    slotsHTML += `<div class="h-16 border-b border-gray-200 relative"></div>`;
                } else if (hour === 19 && vetId === 'sunday') {
                    slotsHTML += `<div class="h-16 bg-gray-50"></div>`;
                } else {
                    slotsHTML += `
                        <div class="h-16 border-b border-gray-200 relative">
                            <p class="text-xs text-center text-gray-400 mt-6">Disponible</p>
                        </div>
                    `;
                }
            }
            
            // Special case for Sunday (closed)
            if (vetId === 'sunday') {
                slotsHTML = Array(12).fill('<div class="h-16 border-b border-gray-200 bg-gray-50"></div>').join('');
            }
            
            grid.innerHTML = `
                <div class="border-r border-gray-200">
                    <div class="h-16 border-b border-gray-200 bg-pawcare-gray flex items-center justify-center">
                        <span class="text-sm text-gray-500">${vet.name}</span>
                    </div>
                    ${slotsHTML}
                </div>
            `;
        }

        function showAppointmentModal(appointmentName) {
            document.getElementById('appointment-modal').classList.remove('hidden');
        }

        // Tooltip functionality
        document.addEventListener('mouseover', function(e) {
            if (e.target.closest('[data-tooltip]')) {
                const tooltip = document.getElementById('tooltip');
                const tooltipText = e.target.closest('[data-tooltip]').getAttribute('data-tooltip');
                tooltip.innerHTML = tooltipText;
                tooltip.classList.remove('hidden');
                
                const rect = e.target.getBoundingClientRect();
                tooltip.style.left = rect.left + 'px';
                tooltip.style.top = (rect.top - tooltip.offsetHeight - 5) + 'px';
            }
        });

        document.addEventListener('mouseout', function(e) {
            if (e.target.closest('[data-tooltip]')) {
                document.getElementById('tooltip').classList.add('hidden');
            }
        });
