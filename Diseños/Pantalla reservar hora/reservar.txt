    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const visitReasonSelect = document.getElementById('visit-reason');
            const otherReasonField = document.getElementById('other-reason-field');
            const additionalInfoTextarea = document.getElementById('additional-info');
            const charCount = document.getElementById('char-count');
            const form = document.getElementById('appointment-form');
            const continueBtn = document.getElementById('continue-btn');
            const btnText = document.getElementById('btn-text');
            const btnIcon = document.getElementById('btn-icon');
            const loadingIcon = document.getElementById('loading-icon');

            visitReasonSelect.addEventListener('change', function() {
                if (this.value === 'otro') {
                    otherReasonField.classList.remove('hidden');
                } else {
                    otherReasonField.classList.add('hidden');
                }
                clearError('visit-reason');
            });

            additionalInfoTextarea.addEventListener('input', function() {
                charCount.textContent = this.value.length;
            });

            document.querySelectorAll('input[name="first-visit"]').forEach(radio => {
                radio.addEventListener('change', function() {
                    clearError('first-visit');
                });
            });

            function clearError(fieldName) {
                const errorElement = document.getElementById(`error-${fieldName}`);
                if (errorElement) {
                    errorElement.classList.add('hidden');
                }
            }

            function showError(fieldName, message) {
                const errorElement = document.getElementById(`error-${fieldName}`);
                if (errorElement) {
                    errorElement.classList.remove('hidden');
                }
            }

            form.addEventListener('submit', function(e) {
                e.preventDefault();
                
                let isValid = true;

                if (!visitReasonSelect.value) {
                    showError('visit-reason');
                    isValid = false;
                }

                const firstVisitRadio = document.querySelector('input[name="first-visit"]:checked');
                if (!firstVisitRadio) {
                    showError('first-visit');
                    isValid = false;
                }

                if (isValid) {
                    btnText.textContent = 'Procesando...';
                    btnIcon.classList.add('hidden');
                    loadingIcon.classList.remove('hidden');
                    continueBtn.disabled = true;

                    setTimeout(() => {
                        console.log('Form submitted successfully');
                        btnText.textContent = 'Continuar';
                        btnIcon.classList.remove('hidden');
                        loadingIcon.classList.add('hidden');
                        continueBtn.disabled = false;
                    }, 2000);
                }
            });
        });
    </script>
